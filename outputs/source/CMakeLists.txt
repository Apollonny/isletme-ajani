cmake_minimum_required(VERSION 3.28)
project(YerelAjan VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../app")
set(AJAN_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../work/deps" CACHE PATH "Pinned dependency source tree")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
add_subdirectory("${AJAN_DEPS}/llama.cpp-0.6.0" llama EXCLUDE_FROM_ALL)
add_subdirectory("${AJAN_DEPS}/whisper.cpp-1.9.5" whisper EXCLUDE_FROM_ALL)
add_library(sqlite STATIC "${AJAN_DEPS}/sqlite-amalgamation-3530000/sqlite3.c")
target_compile_definitions(sqlite PRIVATE SQLITE_ENABLE_FTS5 SQLITE_THREADSAFE=1)
target_include_directories(sqlite PUBLIC "${AJAN_DEPS}/sqlite-amalgamation-3530000")
set(ORT "${AJAN_DEPS}/onnxruntime-win-x64-1.31.0")
add_library(ort SHARED IMPORTED)
set_target_properties(ort PROPERTIES IMPORTED_IMPLIB "${ORT}/lib/onnxruntime.lib" IMPORTED_LOCATION "${ORT}/lib/onnxruntime.dll" INTERFACE_INCLUDE_DIRECTORIES "${ORT}/include")
add_library(ajan STATIC common.cpp store.cpp agent.cpp speech.cpp plugins.cpp doctor.cpp)
target_include_directories(ajan PUBLIC "${AJAN_DEPS}/include" .)
target_compile_definitions(ajan PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ajan PUBLIC sqlite llama whisper ort winmm advapi32 crypt32 bcrypt psapi shell32)
set(SHERPA "${AJAN_DEPS}/sherpa-onnx-v1.13.8-win-x64-shared-MD-Release")
target_include_directories(ajan PRIVATE "${SHERPA}/include")
target_link_libraries(ajan PUBLIC "${SHERPA}/lib/sherpa-onnx-c-api.lib")
configure_file("${SHERPA}/lib/sherpa-onnx-c-api.dll" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/sherpa-onnx-c-api.dll" COPYONLY)
add_executable(YerelAjan main.cpp ui.cpp)
target_link_libraries(YerelAjan PRIVATE ajan)
add_executable(YerelAjanGateway gateway.cpp)
target_link_libraries(YerelAjanGateway PRIVATE ajan ws2_32 winhttp httpapi)
target_link_options(YerelAjan PRIVATE -municode)
target_link_options(YerelAjanGateway PRIVATE -municode)
add_executable(ajan-tests tests.cpp)
target_link_libraries(ajan-tests PRIVATE ajan)
enable_testing()
add_test(NAME agent_safety COMMAND ajan-tests)
foreach(t YerelAjan YerelAjanGateway ajan-tests)
  target_link_options(${t} PRIVATE -static-libgcc -static-libstdc++)
endforeach()
add_executable(YerelAjanSMS sms.cpp)
target_include_directories(YerelAjanSMS PRIVATE "${AJAN_DEPS}/include" "${AJAN_DEPS}/Gammu-1.45.0/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../work/gammubuild/include")
target_link_libraries(YerelAjanSMS PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../work/gammubuild/libgammu/libGammu.dll.a")
target_link_options(YerelAjanSMS PRIVATE -static-libstdc++)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../../work/gammubuild/libgammu/libGammu.dll" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/libGammu.dll" COPYONLY)
add_executable(InventoryPlugin sample_plugin.cpp)
target_include_directories(InventoryPlugin PRIVATE "${AJAN_DEPS}/include")
set_target_properties(InventoryPlugin PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/plugins")
target_link_options(InventoryPlugin PRIVATE -static -static-libstdc++)
configure_file("${ORT}/lib/onnxruntime.dll" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/onnxruntime.dll" COPYONLY)
set(PJLIB_WITH_SSL schannel CACHE STRING "" FORCE)
set(PJMEDIA_WITH_VIDEO OFF CACHE BOOL "" FORCE)
set(PJMEDIA_WITH_WEBRTC_AEC OFF CACHE BOOL "" FORCE)
set(PJMEDIA_WITH_WEBRTC_AEC3 OFF CACHE BOOL "" FORCE)
set(PJ_SKIP_EXPERIMENTAL_NOTICE ON CACHE BOOL "" FORCE)
add_subdirectory("${AJAN_DEPS}/pjproject-2.17" pjsip EXCLUDE_FROM_ALL)
add_executable(YerelAjanSIP sip.cpp)
target_include_directories(YerelAjanSIP PRIVATE "${CMAKE_BINARY_DIR}/pjsip/pjsip/include" "${CMAKE_BINARY_DIR}/pjsip/pjmedia/include")
target_link_libraries(YerelAjanSIP PRIVATE ajan pjsua2)
target_link_options(YerelAjanSIP PRIVATE -municode -static-libstdc++)
add_executable(SipProbe sip_probe.cpp)
target_include_directories(SipProbe PRIVATE "${CMAKE_BINARY_DIR}/pjsip/pjsip/include" "${CMAKE_BINARY_DIR}/pjsip/pjmedia/include")
target_link_libraries(SipProbe PRIVATE ajan pjsua2)
target_link_options(SipProbe PRIVATE -municode -static-libstdc++)
foreach(t YerelAjanSIP SipProbe)
  foreach(d pjlib pjlib-util pjnath pjmedia pjsip)
    target_include_directories(${t} PRIVATE "${AJAN_DEPS}/pjproject-2.17/${d}/include" "${CMAKE_BINARY_DIR}/pjsip/${d}/include")
  endforeach()
endforeach()
